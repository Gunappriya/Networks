#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <time.h>
#include <arpa/inet.h>

#define PORT 8080
#define BUFFER_SIZE 4096
#define FILE_NAME "students.txt"

typedef struct {
    char reg_no[20];
    char name[50];
    char dept[20];
    int semester;
    float marks;
    char grade[5];
    float cgpa;
} Student;

void log_transaction(const char *action) {
    FILE *log = fopen("transactions.log", "a");
    if (log) {
        time_t now = time(NULL);
        char *dt = ctime(&now);
        dt[strlen(dt) - 1] = '\0';
        fprintf(log, "[%s] ACTION: %s\n", dt, action);
        fclose(log);
    }
}

void add_student(Student s) {
    FILE *fp = fopen(FILE_NAME, "a");
    if (fp) {
        fprintf(fp, "%s %s %s %d %.2f %s %.2f\n", s.reg_no, s.name, s.dept, s.semester, s.marks, s.grade, s.cgpa);
        fclose(fp);
    }
}

int search_student(const char *reg_no, char *output) {
    FILE *fp = fopen(FILE_NAME, "r");
    if (!fp) return 0;

    Student s;
    while (fscanf(fp, "%s %s %s %d %f %s %f", s.reg_no, s.name, s.dept, &s.semester, &s.marks, s.grade, &s.cgpa) != EOF) {
        if (strcmp(s.reg_no, reg_no) == 0) {
            sprintf(output, "Reg No: %s | Name: %s | Dept: %s | Sem: %d | Marks: %.2f | Grade: %s | CGPA: %.2f",
                    s.reg_no, s.name, s.dept, s.semester, s.marks, s.grade, s.cgpa);
            fclose(fp);
            return 1;
        }
    }
    fclose(fp);
    return 0;
}

void get_topper(char *output) {
    FILE *fp = fopen(FILE_NAME, "r");
    if (!fp) {
        strcpy(output, "No records found.");
        return;
    }

    Student s, topper;
    float max_cgpa = -1.0;
    int found = 0;

    while (fscanf(fp, "%s %s %s %d %f %s %f", s.reg_no, s.name, s.dept, &s.semester, &s.marks, s.grade, &s.cgpa) != EOF) {
        if (s.cgpa > max_cgpa) {
            max_cgpa = s.cgpa;
            topper = s;
            found = 1;
        }
    }
    fclose(fp);

    if (found) {
        sprintf(output, "CLASS TOPPER -> Reg No: %s | Name: %s | CGPA: %.2f", topper.reg_no, topper.name, topper.cgpa);
    } else {
        strcpy(output, "No records available.");
    }
}

int main() {
    int server_fd, new_socket;
    struct sockaddr_in address;
    int addrlen = sizeof(address);
    char buffer[BUFFER_SIZE];

    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(PORT);

    bind(server_fd, (struct sockaddr *)&address, sizeof(address));
    listen(server_fd, 3);
    printf("Student Result Management Server running on port %d...\n", PORT);

    new_socket = accept(server_fd, (struct sockaddr *)&address, (socklen_t*)&addrlen);

    while (1) {
        memset(buffer, 0, BUFFER_SIZE);
        int bytes_read = read(new_socket, buffer, BUFFER_SIZE);
        if (bytes_read <= 0) break;

        char command[20];
        sscanf(buffer, "%s", command);

        if (strcmp(command, "ADD") == 0) {
            Student s;
            sscanf(buffer, "ADD %s %s %s %d %f %s %f", s.reg_no, s.name, s.dept, &s.semester, &s.marks, s.grade, &s.cgpa);
            add_student(s);
            send(new_socket, "Record added successfully.", 26, 0);
            log_transaction("Added student record.");
        } else if (strcmp(command, "SEARCH") == 0) {
            char reg_no[20], result[BUFFER_SIZE];
            sscanf(buffer, "SEARCH %s", reg_no);
            if (search_student(reg_no, result)) {
                send(new_socket, result, strlen(result), 0);
            } else {
                send(new_socket, "Error: Record not found.", 24, 0);
            }
            log_transaction("Searched student record.");
        } else if (strcmp(command, "TOPPER") == 0) {
            char result[BUFFER_SIZE];
            get_topper(result);
            send(new_socket, result, strlen(result), 0);
            log_transaction("Retrieved class topper.");
        } else {
            send(new_socket, "Error: Invalid operation.", 25, 0);
            log_transaction("Invalid request received.");
        }
    }

    close(new_socket);
    close(server_fd);
    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080
#define BUFFER_SIZE 4096

int main() {
    int sock = 0;
    struct sockaddr_in serv_addr;
    char buffer[BUFFER_SIZE] = {0};

    sock = socket(AF_INET, SOCK_STREAM, 0);
    serv_addr.sin_family = AF_INET;
    serv_addr.sin_port = htons(PORT);
    inet_pton(AF_INET, "127.0.0.1", &serv_addr.sin_addr);

    if (connect(sock, (struct sockaddr *)&serv_addr, sizeof(serv_addr)) < 0) {
        printf("Connection Failed\n");
        return -1;
    }

    int choice;
    while (1) {
        printf("\n=== STUDENT RESULT MANAGEMENT ===\n");
        printf("1. Add Student\n2. Search Student\n3. Display Class Topper\n4. Exit\nChoice: ");
        scanf("%d", &choice);

        if (choice == 1) {
            char reg[20], name[50], dept[20], grade[5];
            int sem;
            float marks, cgpa;

            printf("Enter Reg No, Name, Dept, Sem, Marks, Grade, CGPA:\n");
            scanf("%s %s %s %d %f %s %f", reg, name, dept, &sem, &marks, grade, &cgpa);

            sprintf(buffer, "ADD %s %s %s %d %.2f %s %.2f", reg, name, dept, sem, marks, grade, cgpa);
            send(sock, buffer, strlen(buffer), 0);

            memset(buffer, 0, BUFFER_SIZE);
            read(sock, buffer, BUFFER_SIZE);
            printf("Server Response: %s\n", buffer);
        } else if (choice == 2) {
            char reg[20];
            printf("Enter Reg No to Search: ");
            scanf("%s", reg);

            sprintf(buffer, "SEARCH %s", reg);
            send(sock, buffer, strlen(buffer), 0);

            memset(buffer, 0, BUFFER_SIZE);
            read(sock, buffer, BUFFER_SIZE);
            printf("Server Response: %s\n", buffer);
        } else if (choice == 3) {
            send(sock, "TOPPER", 6, 0);
            memset(buffer, 0, BUFFER_SIZE);
            read(sock, buffer, BUFFER_SIZE);
            printf("Server Response: %s\n", buffer);
        } else if (choice == 4) {
            break;
        }
    }

    close(sock);
    return 0;
}
